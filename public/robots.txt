# NX036 — Política de rastreo.
# Objetivo: permitir que los buscadores tradicionales (Google/Bing) indexen la tienda
# para traer clientes, pero BLOQUEAR los crawlers de IA / entrenamiento / scrapers que
# se usan para clonar contenido y datos.

Sitemap: https://nx036.com/sitemap.xml
